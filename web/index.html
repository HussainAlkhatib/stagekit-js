stagekit-js is a catalog of 4,658 small, pure string‑transformation stages. Each one does exactly one thing. Chain them into a pipeline and run them from code, from the command line — or try it live below.

npm install stagekit-js

      What's inside
      
        
      4,658 stages across 16 families. Every name and behaviour is unique - enforced by the test suite.
      
      
      
      
      
    
      

    
      
      
      Use it from code
      
      
      
        
      
          
      Compose a pipeline
      
          
      const { registry, createPipeline } = require('stagekit-js');

const pipe = createPipeline([
  registry.findByName('Trim lines')[0],
  registry.findByName('Slug with "-"')[0],
]);

pipe('  Hello   World  ');
// -> 'hello-world'
      
        
      
        
      
          
      Search the registry
      
          
      registry.size;            // 4658
registry.search('caesar'); // every caesar stage
registry.get('mod-0002'); // one exact stage

registry.search('slug')[0].run('My Post');
// -> 'my-post'
